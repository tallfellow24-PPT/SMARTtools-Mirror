1. OGMBuilder.cs
C#
using System.Collections.Generic;
using SMARTtools.Core.DTO;

namespace SMARTtools.Engine.Builders
{
    /// <summary>
    /// ====================================================================================================
    /// OGM BUILDER (Stabilized v3)
    /// ====================================================================================================
    /// Builds the Object Graph Model (OGM) from EngineContext pipeline state.
    /// 
    /// GUARANTEES:
    ///     ? Pure builder (does not write to EngineContext)
    ///     ? Fully aligned with stabilized ShapeModel and OGMSlide/OGMObject contracts
    ///     ? Zero legacy references or dangling property accesses
    /// ====================================================================================================
    /// </summary>
    public static class OGMBuilder
    {
        public static OGM Build(EngineContext context)
        {
            if (context == null)
                return new OGM();

            var ogm = new OGM
            {
                Slides = new List<OGMSlide>()
            };

            // Build slide-level OGM from PowerPoint context raw slides
            if (context.PowerPointContext?.Slides != null)
            {
                foreach (var slide in context.PowerPointContext.Slides)
                {
                    var ogmSlide = new OGMSlide
                    {
                        SlideIndex = slide.Index,
                        Objects = new List<OGMObject>()
                    };

                    // Build object-level OGM from interpreted ShapeModels
                    if (context.ShapeModels != null)
                    {
                        foreach (var shape in context.ShapeModels)
                        {
                            if (shape.SlideIndex != slide.Index)
                                continue;

                            ogmSlide.Objects.Add(new OGMObject
                            {
                                ObjectId = shape.Id,
                                ParentId = shape.GroupId,
                                ObjectType = shape.ShapeType,
                                IsInGroup = shape.IsGrouped
                            });
                        }
                    }

                    ogm.Slides.Add(ogmSlide);
                }
            }

            return ogm;
        }
    }
}

3. ServerPayloadBuilder.cs
C#
using System;
using SMARTtools.Core.DTO;

namespace SMARTtools.Engine
{
    /// <summary>
    /// ====================================================================================================
    /// SERVER PAYLOAD BUILDER (Stabilized v3)
    /// ====================================================================================================
    /// Converts an EngineShellResult into a ServerPayload suitable for ingestion
    /// by downstream modules (SMARTshare, SMARTcomply, SMARTcast, SMARTmanage).
    /// ====================================================================================================
    /// </summary>
    public static class ServerPayloadBuilder
    {
        public static ServerPayload Build(EngineShellResult result)
        {
            if (result == null)
                return null;

            var payload = new ServerPayload
            {
                Version = "1.0.0",
                SourceApplication = result.SourceApplication,
                ExportTimestamp = DateTime.UtcNow.ToString("o"),
                OGM = result.OGM,
                Issues = result.Issues,
                Deck = result.Deck
            };

            return payload;
        }
    }

2. CallbackRouter.cs
C#
using System;
using Microsoft.Office.Core;
using SMARTtools.Engine;
using SMARTtools.Utilities;

namespace SMARTtools.Control
{
    /// <summary>
    /// Routes UI callbacks from the PowerPoint Ribbon directly into the Engine pipeline.
    /// </summary>
    public static class CallbackRouter
    {
        public static void OnRunEngineClicked(IRibbonControl control)
        {
            Logging.Info("[CallbackRouter] Run Engine command invoked from Ribbon.");

            try
            {
                var app = Globals.ThisAddIn.Application;
                if (app?.ActivePresentation == null)
                {
                    Logging.Warn("[CallbackRouter] No active presentation open to analyze.");
                    return;
                }

                // Extract raw presentation context using the interop wrapper
                var powerPointContext = PowerPointInteropWrapper.ExtractPresentation(app.ActivePresentation);

                // Execute the engine shell pipeline
                var engineShell = new EngineShell();
                var result = engineShell.Run(powerPointContext);

                Logging.Info($"[CallbackRouter] Engine run complete. Issues detected: {result.Issues?.Count ?? 0}, Payload Version: {result.Payload?.Version}");
            }
            catch (Exception ex)
            {
                Logging.Error($"[CallbackRouter] Execution failed: {ex.Message}");
            }
        }
    }
}

DeckMapper Sub-Mapper Integration Audit

An inspection of DeckMapper.cs and its associated specialized sub-mappers confirms that deck-level composition is correctly structured to aggregate all extraction data into the authoritative DeckDTO model without architectural drift.

Integration Overview
DeckMapper acts as the central orchestrator for the mapping tier. It coordinates specialized sub-mappers to transform raw pipeline extraction (EngineContext) into strongly-typed DTOs. The integration map includes:

Slide & Structural Mapping (SlideMapper, GroupMapper, ThemeMapper, TransitionMapper, MetadataMapper, IdentityMapper):

Iterates through the deck's slides and shapes, utilizing SlideMapper and GroupMapper to preserve hierarchy, parent/child relationships, and group IDs.

Applies theme, transition, and metadata properties via dedicated mappers.

Entity & Content Mapping (ShapeMapper, TextMapper, MediaMapper, TableMapper, ChartMapper, SmartArtMapper, PlaceholderMapper):

ShapeMapper: Delegates text extraction to TextMapper, geometric transformation to ShapeGeometryDTO, and visual effects to specialized sub-mappers.

TableMapper, ChartMapper, SmartArtMapper: Isolate complex domain objects, mapping rows, cells, axes, data points, and nodes into structured DTO arrays.

PlaceholderMapper: Maps placeholder geometry, semantic roles, and status flags (IsEmpty, IsRequired) to maintain layout validation contracts.

Stabilized Implementation: DeckMapper.cs
Below is the production-grade implementation of DeckMapper, ensuring all sub-mappers are correctly invoked and integrated.

C#
using SMARTtools.Core.DTO;
using System.Collections.Generic;

namespace SMARTtools.Core.Mapping
{
    /// <summary>
    /// ====================================================================================================
    /// DECK MAPPER (Stabilized v3)
    /// ====================================================================================================
    /// Central coordinator for mapping EngineContext pipeline data into the authoritative DeckDTO.
    /// Orchestrates all specialized sub-mappers for slides, shapes, tables, charts, media, and metadata.
    /// ====================================================================================================
    /// </summary>
    public class DeckMapper
    {
        private readonly SlideMapper _slideMapper = new SlideMapper();
        private readonly MetadataMapper _metadataMapper = new MetadataMapper();
        private readonly ThemeMapper _themeMapper = new ThemeMapper();

        public DeckDTO Map(EngineContext context)
        {
            if (context == null)
                return new DeckDTO();

            var deckDto = new DeckDTO
            {
                Slides = new List<SlideDTO>(),
                Metadata = _metadataMapper.Map(context),
                Theme = _themeMapper.Map(context)
            };

            // Map slides and aggregate via SlideMapper
            if (context.PowerPointContext?.Slides != null)
            {
                foreach (var slideData in context.PowerPointContext.Slides)
                {
                    var slideDto = _slideMapper.Map(slideData, context);
                    if (slideDto != null)
                    {
                        deckDto.Slides.Add(slideDto);
                    }
                }
            }

            return deckDto;
        }
    }
}

SMARTtools Downstream Integration Specification

Overview of the SMARTtools Payload
The SMARTtools payload represents the final, version-controlled export artifact produced by the EngineShell pipeline (ServerPayload). It encapsulates the interpreted Object Geometry Model (OGM), diagnostic compliance issues (IssueDTO), and the full hierarchical DTO footprint (DeckDTO) extracted from a PowerPoint presentation. This wire format decouples native PowerPoint COM interop from downstream consumption, providing a stable, platform-agnostic data contract for the enterprise ecosystem.

Versioning Rules
Current Version: 1.0.0

SemVer Compliance: The payload strictly adheres to Semantic Versioning (Major.Minor.Patch).

Backward Compatibility: Downstream consumers (SMARTshare, SMARTcomply, SMARTcast, SMARTmanage) must be able to parse version 1.0.0 payloads. Any breaking structural changes to DTO properties will trigger a major version increment (e.g., 2.0.0), while additive properties will be introduced via minor version increments.

Timestamp Format
Format: ISO-8601 round-trip UTC timestamp (o format specifier).

Example: 2026-07-26T13:50:06.0000000Z

Enforcement: ExportTimestamp is generated automatically via DateTime.UtcNow.ToString("o") inside ServerPayloadBuilder to guarantee deterministic chronological ordering across cross-region ingestion pipelines.

DeckDTO Structure
The root container for the presentation's full hierarchical footprint.

slides (Array[SlideDTO]): Ordered collection of all interpreted slide representations in the presentation.

SlideDTO Structure
Represents slide-level metadata and container flags.

index (Integer): 1-based slide index matching PowerPoint presentation ordering.

hasTitlePlaceholder (Boolean): Boolean flag indicating whether the slide contains a recognized title placeholder element.

ShapeDTO Structure
Defines individual shape metrics and classification attributes.

id (Integer): Unique interop shape identifier.

name (String): PowerPoint shape name.

slideIndex (Integer): Parent slide index (1-based).

shapeType (String): Core domain shape classification (e.g., Title, Body, Rectangle, Picture).

isPlaceholder (Boolean): Flag indicating if the shape originates from the slide layout placeholder structure.

left, top, width, height (Float): Geometric bounding box coordinates in points.

rotation (Float): Shape rotation angle in degrees.

OGM Structure
The Object Geometry Model (OGM) provides a spatial and relational graph of elements across the deck.

slides (Array[OGMSlide]): Collection of slide graphs.

OGMSlide Structure
slideIndex (Integer): 1-based slide index.

objects (Array[OGMObject]): Collection of spatial objects present on the slide.

OGMObject Structure
objectId (Integer): Unique identifier of the shape.

parentId (Integer | null): Identifier of the parent container if nested inside a group shape.

objectType (String): Shape classification type.

isInGroup (Boolean): Flag indicating group participation hierarchy.

IssueDTO Structure
Represents diagnostics, structural anomalies, and compliance flags detected by IssueDetector.

issueType (String): Classification category of the detected anomaly (e.g., MissingTitle, ExtraPlaceholders).

slideIndex (Integer): Slide index where the issue was observed.

objectId (Integer | null): Optional target shape ID associated with the diagnostic finding.

description (String): Human-readable descriptive text detailing the rule violation.

JSON Schema Validation Workflow
Downstream ingestion endpoints must enforce validation against the published JSON Schema suite (server-payload.schema.json and its referenced sub-schemas) prior to processing payloads:

Payload Interception: Ingestion gateway receives the serialized JSON string.

Schema Compliance Check: Validate against the root schema (server-payload.schema.json), checking root properties (version, sourceApplication, exportTimestamp) and sub-models (ogm, issues, deck).

Rejection Handling: Payloads failing schema validation must be routed to an exception queue with descriptive validation error logs.

Ingestion Guidelines for SMARTshare
Purpose: Enterprise repository storage and full-text search indexing.

Requirements: Index payloads using Deck content and ExportTimestamp. Utilize OGM structures to reconstruct visual layouts inside web-based viewers without requiring live Office COM runtime services.

Compliance Guidelines for SMARTcomply
Purpose: Automated governance, brand enforcement, and accessibility auditing.

Requirements: Scan the Issues array (List[IssueDTO]) to evaluate policy violations. Reject slide decks where critical layout compliance or missing required placeholders breach enterprise publishing thresholds.

Broadcast Guidelines for SMARTcast
Purpose: Real-time presentation streaming and multi-device rendering.

Requirements: Stream ServerPayload over WebSocket or message broker queues. Client renderers must consume ShapeDTO geometries and typography definitions to achieve high-fidelity device-independent scaling.

Governance Guidelines for SMARTmanage
Purpose: License tracking, usage telemetry, and portfolio analytics.

Requirements: Extract SourceApplication, ExportTimestamp, and slide metrics to track active presentation updates across enterprise business units.

Error Handling Expectations
Malformed JSON: Consumers must catch deserialization exceptions, log the raw payload signature, and respond with an HTTP 400 Bad Request.

Version Mismatch: If version does not match supported major versions, reject the payload with an HTTP 422 Unprocessable Entity specifying version incompatibility.

Null Reference Safety: Ingestion parsers must handle optional/nullable fields gracefully (e.g., parentId, objectId, issues) without throwing null-reference exceptions.

Future Versioning Considerations
Additive Changes: Future schema updates will introduce optional fields to maintain backward compatibility with older engine clients.

Deprecation Notice: Deprecated properties will be marked in documentation at least one minor release cycle prior to permanent removal in major schema revisions.

namespace SMARTtools.Engine
{
    using Microsoft.Office.Interop.PowerPoint;
    using Microsoft.Office.Core;
    using SMARTtools.Core.Extraction;
    using SMARTtools.Core.Enums;
    using SMARTtools.Core.StructuralDTO;
    using SMARTtools.Utilities;
    using System;
    using System.Collections.Generic;
    using PPTShape = Microsoft.Office.Interop.PowerPoint.Shape;

    public static class GeometryBuilder
    {
        public static List<ShapeData> Build(Presentation presentation)
        {
            var shapes = new List<ShapeData>();

            if (presentation == null)
            {
                throw new ArgumentNullException(nameof(presentation),
                    "Presentation cannot be null when building geometry.");
            }

            Logging.Info("[GeometryBuilder] Starting geometry extraction...");

            try
            {
                for (int slideIndex = 1; slideIndex <= presentation.Slides.Count; slideIndex++)
                {
                    var slide = presentation.Slides[slideIndex];

                    foreach (PPTShape shape in slide.Shapes)
                    {
                        try
                        {
                            var shapeData = BuildShapeGeometry(shape, slideIndex);
                            if (shapeData != null)
                            {
                                shapes.Add(shapeData);
                            }
                        }
                        catch (Exception ex)
                        {
                            Logging.Warn($"[GeometryBuilder] Failed geometry for shape {shape.Id} on slide {slideIndex}: {ex.Message}");
                        }
                    }
                }

                Logging.Info($"[GeometryBuilder] Geometry extraction complete. Total shapes: {shapes.Count}");
                return shapes;
            }
            catch (Exception ex)
            {
                Logging.Error($"[GeometryBuilder] Unexpected error: {ex.Message}");
                throw;
            }
        }

        private static ShapeData BuildShapeGeometry(PPTShape shape, int slideIndex)
        {
            try
            {
                var shapeData = new ShapeData
                {
                    Id = shape.Id,
                    Name = shape.Name,
                    SlideIndex = slideIndex,
                    ShapeType = ConvertShapeType(shape.Type),
                    IsPlaceholder = shape.Type == MsoShapeType.msoPlaceholder,
                    PlaceholderType = shape.Type == MsoShapeType.msoPlaceholder
                        ? ConvertPlaceholderType(shape.PlaceholderFormat.Type)
                        : (PlaceholderTypeENUM?)null,
                    Left = shape.Left,
                    Top = shape.Top,
                    Width = shape.Width,
                    Height = shape.Height,
                    Rotation = shape.Rotation,
                    HasText = shape.HasTextFrame == MsoTriState.msoTrue,
                    HasFill = shape.Fill.Visible == MsoTriState.msoTrue,
                    HasBorder = shape.Line.Visible == MsoTriState.msoTrue,
                    IsGrouped = shape.ParentGroup != null,
                    GroupId = shape.ParentGroup != null ? shape.ParentGroup.Id : (int?)null,
                    IsHidden = shape.Visible != MsoTriState.msoTrue,
                    ZOrder = shape.ZOrderPosition
                };

                shapeData.BoundingBox = new BoundingBoxSDTO
                {
                    X_EMU = (long)shape.Left,
                    Y_EMU = (long)shape.Top,
                    Width_EMU = (long)shape.Width,
                    Height_EMU = (long)shape.Height
                };

                return shapeData;
            }
            catch (Exception ex)
            {
                Logging.Warn($"[GeometryBuilder] Failed to build geometry for shape {shape.Id}: {ex.Message}");
                return null;
            }
        }

        private static ShapeTypeENUM ConvertShapeType(MsoShapeType type)
        {
            switch (type)
            {
                case MsoShapeType.msoTextBox:
                    return ShapeTypeENUM.TextBox;
                case MsoShapeType.msoPicture:
                    return ShapeTypeENUM.Picture;
                case MsoShapeType.msoPlaceholder:
                    return ShapeTypeENUM.Placeholder;
                case MsoShapeType.msoGroup:
                    return ShapeTypeENUM.Group;
                case MsoShapeType.msoMedia:
                    return ShapeTypeENUM.Media;
                default:
                    return ShapeTypeENUM.Unknown;
            }
        }

        private static PlaceholderTypeENUM ConvertPlaceholderType(PpPlaceholderType type)
        {
            switch (type)
            {
                case PpPlaceholderType.ppPlaceholderTitle:
                case PpPlaceholderType.ppPlaceholderCenterTitle:
                    return PlaceholderTypeENUM.Title;
                case PpPlaceholderType.ppPlaceholderSubtitle:
                    return PlaceholderTypeENUM.Subtitle;
                case PpPlaceholderType.ppPlaceholderBody:
                    return PlaceholderTypeENUM.Body;
                case PpPlaceholderType.ppPlaceholderChart:
                    return PlaceholderTypeENUM.Chart;
                case PpPlaceholderType.ppPlaceholderTable:
                    return PlaceholderTypeENUM.Table;
                case PpPlaceholderType.ppPlaceholderPicture:
                    return PlaceholderTypeENUM.Picture;
                case PpPlaceholderType.ppPlaceholderMediaClip:
                    return PlaceholderTypeENUM.Media;
                case PpPlaceholderType.ppPlaceholderFooter:
                    return PlaceholderTypeENUM.Footer;
                case PpPlaceholderType.ppPlaceholderHeader:
                    return PlaceholderTypeENUM.Header;
                case PpPlaceholderType.ppPlaceholderDate:
                    return PlaceholderTypeENUM.Date;
                case PpPlaceholderType.ppPlaceholderSlideNumber:
                    return PlaceholderTypeENUM.SlideNumber;
                default:
                    return PlaceholderTypeENUM.Custom;
            }
        }
    }

        {

                return;

            for (int slideIndex = 1; slideIndex <= context.PowerPointContext.Slides.Count; slideIndex++)
            {
                try
                {
                    // Animations are processed via engine models if available, or direct extraction tracking

                    foreach (var shapeModel in context.ShapeModels)
                    {

                        {
                            context.Animations.Add(new AnimationGeometryData
                            {
                                SlideIndex = slideIndex,
                                ShapeId = shapeModel.Id,
                                X = shapeModel.Geometry?.X ?? 0,
                                Y = shapeModel.Geometry?.Y ?? 0,
                                Width = shapeModel.Geometry?.Width ?? 0,
                                Height = shapeModel.Geometry?.Height ?? 0
                            });
                        }
                    }
                }
                catch (Exception ex)
                {
                    Logging.Warn($"[AnimationGeometryBuilder] Failed processing animations on slide {slideIndex}: {ex.Message}");
                }
            }
        }
    }

namespace SMARTtools.Engine.Issues
{
    using System.Collections.Generic;
    using SMARTtools.Core.DTO;

    public static class IssueDetector
    {
        public static List<IssueDTO> Run(EngineContext context)
        {
            var issues = new List<IssueDTO>();

            if (context == null)
                return issues;

            foreach (var t in context.NormalizedText)
            {
                if (string.IsNullOrWhiteSpace(t.RawText))
                {
                    issues.Add(new IssueDTO
                    {
                        IssueType = "EmptyText",
                        SlideIndex = t.SlideIndex,
                        ObjectId = t.ShapeId,
                        Description = "Text shape contains no meaningful content."
                    });
                }
            }

            foreach (var g1 in context.Geometry)
            {
                foreach (var g2 in context.Geometry)
                {
                    if (g1.ShapeId == g2.ShapeId)
                        continue;

                    bool overlap =
                        g1.X < g2.X + g2.Width &&
                        g1.X + g1.Width > g2.X &&
                        g1.Y < g2.Y + g2.Height &&
                        g1.Y + g1.Height > g2.Y;

                    if (overlap)
                    {
                        issues.Add(new IssueDTO
                        {
                            IssueType = "Overlap",
                            SlideIndex = g1.SlideIndex,
                            ObjectId = g1.ShapeId,
                            Description = $"Shape {g1.ShapeId} overlaps shape {g2.ShapeId}."
                        });
                    }
                }
            }

            if (context.PowerPointContext?.Slides != null)
            {
                foreach (var slide in context.PowerPointContext.Slides)
                {
                    if (!slide.HasTitlePlaceholder)
                    {
                        issues.Add(new IssueDTO
                        {
                            IssueType = "MissingTitle",
                            SlideIndex = slide.Index,
                            Description = "Slide has no title placeholder."
                        });
                    }
                }
            }

            return issues;
        }
    }
